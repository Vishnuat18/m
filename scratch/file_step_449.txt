// 3. COILS & BINDING GENERATION

function renderSpiralCoils() {
  coilsContainer.innerHTML = "";
  const numCoils = 19;
  for (let i = 0; i < numCoils; i++) {
    const coil = document.createElement("div");
    coil.className = "metal-coil";
    coilsContainer.appendChild(coil);
  }
}

// ==========================================================================
// 4. ELEMENT FACTORY (Render Photos, Stickers, Typewriters, Notes)
// ==========================================================================
function createScrapbookElement(elConfig) {
  const wrap = document.createElement("div");
  wrap.className = `sb-element ${elConfig.anim || 'pop-in'}`;
  wrap.style.setProperty("--rot", `${elConfig.rot || 0}deg`);
  wrap.style.setProperty("--delay", `${elConfig.delay || 0}ms`);
  wrap.style.setProperty("--z", elConfig.z || 10);
  wrap.style.left = `${elConfig.x}%`;
  wrap.style.top = `${elConfig.y}%`;
  wrap.style.width = `${elConfig.w}%`;

  if (elConfig.h) wrap.style.height = `${elConfig.h}%`;

  // Render by element type
  switch (elConfig.type) {
    case "sticker": {
      const img = document.createElement("img");
      img.className = "sticker-img";
      if (elConfig.idle) img.classList.add(`idle-${elConfig.idle}`);
      img.src = elConfig.src;
      img.alt = "Sticker";
      img.loading = "lazy";
      img.onerror = () => handleImageError(img, elConfig.src);
      wrap.appendChild(img);
      break;
    }

    case "camera-viewfinder": {
      const cameraWrap = document.createElement("div");
      cameraWrap.className = "camera-viewfinder-wrap";
      
      const screenImg = document.createElement("img");
      screenImg.className = "camera-screen-photo";
      screenImg.src = elConfig.screenSrc;
      screenImg.alt = "Viewfinder Memory";
      screenImg.onerror = () => handleImageError(screenImg, elConfig.screenSrc);

      const camSticker = document.createElement("img");
      camSticker.className = "sticker-img camera-sticker-layer";
      camSticker.src = elConfig.cameraSrc;
      camSticker.alt = "Camera";
      camSticker.onerror = () => handleImageError(camSticker, elConfig.cameraSrc);

      cameraWrap.appendChild(screenImg);
      cameraWrap.appendChild(camSticker);
      wrap.appendChild(cameraWrap);
      break;
    }

    case "polaroid": {
      const frame = document.createElement("div");
      frame.className = "polaroid-frame";
      
      if (elConfig.hasTape) {
        const tape = document.createElement("div");
        tape.className = "washi-tape-strip";
        frame.appendChild(tape);
      }

      const img = document.createElement("img");
      img.className = "polaroid-img";
      img.src = elConfig.src;
      img.alt = "Polaroid Photo";
      img.onerror = () => handleImageError(img, elConfig.src);
      frame.appendChild(img);

      if (elConfig.caption) {
        const cap = document.createElement("div");
        cap.className = "polaroid-caption";
        cap.textContent = elConfig.caption;
        frame.appendChild(cap);
      }
      wrap.appendChild(frame);
      break;
    }

    case "photo-strip": {
      const stripFrame = document.createElement("div");
      stripFrame.className = "photo-strip-frame";
      
      if (elConfig.hasTape) {
        const tape = document.createElement("div");
        tape.className = "washi-tape-strip";
        stripFrame.appendChild(tape);
      }

      (elConfig.photos || []).forEach(pSrc => {
        const img = document.createElement("img");
        img.className = "strip-photo";
        img.src = pSrc;
        img.alt = "Photo Strip Item";
        img.onerror = () => handleImageError(img, pSrc);
        stripFrame.appendChild(img);
      });
      wrap.appendChild(stripFrame);
      break;
    }

    case "pink-frame-photo": {
      const frame = document.createElement("div");
      frame.className = "gingham-photo-frame";
      const img = document.createElement("img");
      img.src = elConfig.src;
      img.alt = "Framed Memory";
      img.onerror = () => handleImageError(img, elConfig.src);
      frame.appendChild(img);
      wrap.appendChild(frame);
      break;
    }

    case "ransom-title": {
      const titleWrap = document.createElement("div");
      titleWrap.className = "ransom-title";
      const letters = elConfig.text.split("");
      
      // Random rotation and vintage paper style for each letter
      const bgColors = ["#f7efe1", "#f4b6c8", "#ede5d2", "#ffffff", "#c0172b", "#16131d", "#f8e3a1"];
      const textColors = ["#121212", "#121212", "#121212", "#121212", "#ffffff", "#f7f1e8", "#121212"];
      
      letters.forEach((char, idx) => {
        const span = document.createElement("span");
        span.className = "ransom-letter";
        span.textContent = char;
        const cIdx = idx % bgColors.length;
        span.style.background = bgColors[cIdx];
        span.style.color = textColors[cIdx];
        const rot = (idx % 2 === 0 ? 1 : -1) * (2 + (idx * 2) % 6);
        span.style.setProperty("--lrot", `${rot}deg`);
        titleWrap.appendChild(span);
      });
      wrap.appendChild(titleWrap);
      break;
    }

    case "quote-strip": {
      const strip = document.createElement("div");
      strip.className = "quote-strip";
      strip.dataset.rawText = elConfig.text;
      strip.textContent = ""; // filled via typewriter animation
      wrap.appendChild(strip);
      break;
    }

    case "handwritten-note": {
      const note = document.createElement("div");
      note.className = "handwritten-note";
      note.innerHTML = elConfig.text.replace(/\n/g, "<br>");
      wrap.appendChild(note);
      break;
    }

    case "tag-badge": {
      const tag = document.createElement("div");
      tag.className = "tag-badge";
      tag.textContent = elConfig.text;
      wrap.appendChild(tag);
      break;
    }

    case "texture": {
      const tex = document.createElement("div");
      tex.className = elConfig.subType || "texture-overlay";
      tex.style.backgroundImage = `url('${elConfig.src}')`;
      wrap.appendChild(tex);
      break;
    }

    case "tap-hint": {
      const hint = document.createElement("div");
      hint.className = "tap-to-open";
      hint.innerHTML = `<span style="color:var(--cherry);">♥</span> ${elConfig.text}`;
      hint.addEventListener("click", () => flipNext());
      wrap.appendChild(hint);
      break;
    }

    case "end-title": {
      const h = document.createElement("h1");
      h.className = "end-title";
      h.textContent = elConfig.text;
      wrap.appendChild(h);
      break;
    }

    case "end-footer": {
      const f = document.createElement("p");
      f.className = "end-footer-text";
      f.textContent = elConfig.text;
      wrap.appendChild(f);
      break;
    }

    case "restart-btn": {
      const btn = document.createElement("button");
      btn.className = "restart-btn";
      btn.textContent = elConfig.text;
      btn.addEventListener("click", () => goToPage(0));
      wrap.appendChild(btn);
      break;
    }
  }

  return wrap;
}

// Fallback pretty placeholder if an image fails to load
function handleImageError(imgEl, filename) {
  const box = document.createElement("div");
  box.className = "img-placeholder-box";
  box.style.width = "100%";
  box.style.minHeight = "70px";
  const shortName = filename.split("/").pop();
  box.innerHTML = `<span>📷 [ ${shortName} ]</span>`;
  if (imgEl.parentNode) {
    imgEl.parentNode.replaceChild(box, imgEl);
  }
}

// ==========================================================================
// 5. BOOK BUILDER & RENDER ENGINE
// ==========================================================================
function buildBook() {
  bookEl.innerHTML = "";
  
  // Update header / indicator info
  totalPagesNumEl.textContent = totalPages;
  coverCaptionEl.textContent = BOOK.coverCaption;

  BOOK.pages.forEach((p, idx) => {
    const leaf = document.createElement("div");
    leaf.className = "page-leaf";
    leaf.id = `leaf-${idx}`;
    leaf.dataset.pageIndex = idx;
    
    // Reverse z-index stacking so earlier pages sit on top
    leaf.style.zIndex = totalPages - idx;

    // Front Face (Scrapbook Page Content)
    const frontFace = document.createElement("div");
    frontFace.className = `page-face front ${p.background || 'bg-paper-cream'}`;

    // Moving Shadow Overlay for 3D turn
    const shadowOverlay = document.createElement("div");
    shadowOverlay.className = "page-shadow-overlay";
    frontFace.appendChild(shadowOverlay);

    // Populate all elements on this page
    p.elements.forEach(elDef => {
      const elNode = createScrapbookElement(elDef);
      frontFace.appendChild(elNode);
    });

    // Back Face (Blank aged paper texture for realistic double-sided sheet)
    const backFace = document.createElement("div");
    backFace.className = "page-face back";
    if (idx === 0) {
      backFace.style.backgroundColor = "#1a1622"; // inner cover
    }

    leaf.appendChild(frontFace);
    leaf.appendChild(backFace);
    bookEl.appendChild(leaf);
  });

  // Activate initial cover page
  updatePageState(0);
}

// ==========================================================================
// 6. 3D PAGE TURN PHYSICS & STATE MACHINE
// ==========================================================================
function updatePageState(newIndex) {
  currentPage = newIndex;
  curPageNumEl.textContent = currentPage + 1;

  // Toggle caption bar text based on cover vs inside
  if (currentPage === 0) {
    coverCaptionEl.textContent = BOOK.coverCaption;
    coverCaptionEl.style.opacity = "1";
  } else {
    coverCaptionEl.textContent = BOOK.pages[currentPage].name || `Page ${currentPage}`;
    coverCaptionEl.style.opacity = "0.75";
  }

  // Update classes and 3D transforms for every leaf
  const leaves = document.querySelectorAll(".page-leaf");
  leaves.forEach((leaf, idx) => {
    if (idx < currentPage) {
      // Already turned pages: rotated -180deg, stacked in order of turn
      leaf.classList.add("turned");
      leaf.classList.remove("active");
      leaf.style.zIndex = idx + 1;
    } else if (idx === currentPage) {
      // Currently active page: flat 0deg, on top of remaining pages
      leaf.classList.remove("turned");
      leaf.classList.add("active");
      leaf.style.zIndex = totalPages + 10;
      
      // Trigger typewriter effect for quote strips
      triggerPageTypewriters(leaf);
    } else {
      // Future pages: flat 0deg, stacked normally
      leaf.classList.remove("turned");
      leaf.classList.remove("active");
      leaf.style.zIndex = totalPages - idx;
    }
  });
}

function flipNext() {
  if (isFlipping || currentPage >= totalPages - 1) return;
  isFlipping = true;

  const currentLeaf = document.getElementById(`leaf-${currentPage}`);
  if (currentLeaf) {
    currentLeaf.classList.add("turning");
  }

  // Auto start music on first user interaction if not playing
  if (!audioPlayed) {
    startMusic();
  }

  updatePageState(currentPage + 1);

  setTimeout(() => {
    if (currentLeaf) currentLeaf.classList.remove("turning");
    isFlipping = false;
  }, 920);
}

function flipPrev() {
  if (isFlipping || currentPage <= 0) return;
  isFlipping = true;

  const prevLeaf = document.getElementById(`leaf-${currentPage - 1}`);
  if (prevLeaf) {
    prevLeaf.classList.add("turning");
    prevLeaf.style.zIndex = totalPages + 15; // bring turned leaf above during turn back
  }

  updatePageState(currentPage - 1);

  setTimeout(() => {
    if (prevLeaf) prevLeaf.classList.remove("turning");
    isFlipping = false;
  }, 920);
}

function goToPage(targetIndex) {
  if (isFlipping) return;
  targetIndex = Math.max(0, Math.min(totalPages - 1, targetIndex));
  updatePageState(targetIndex);
}

// ==========================================================================
// 7. TYPEWRITER ANIMATION ENGINE FOR QUOTE STRIPS
// ==========================================================================
function triggerPageTypewriters(pageLeafEl) {
  const quoteStrips = pageLeafEl.querySelectorAll(".quote-strip");
  quoteStrips.forEach((strip, sIdx) => {
    const raw = strip.dataset.rawText || "";
    strip.textContent = "";
    strip.classList.remove("done");
    
    let charIdx = 0;
    // Delay start until after stickers pop in
    const startDelay = 500 + sIdx * 250;
    
    setTimeout(() => {
      const interval = setInterval(() => {
        if (charIdx < raw.length) {
          strip.textContent += raw[charIdx];
          charIdx++;
        } else {
          clearInterval(interval);
          strip.classList.add("done");
        }
      }, 35);
    }, startDelay);
  });
}

// ==========================================================================
// 8. BACKGROUND FLOATING PARTICLES (Stars & Petals)
// ==========================================================================
function initAmbientCanvas() {
  const canvas = document.getElementById("ambient-canvas");
  const ctx = canvas.getContext("2d");
  let width, height;
  let particles = [];

  function resize() {
    width = canvas.width = window.innerWidth;
    height = canvas.height = window.innerHeight;
  }
  window.addEventListener("resize", resize);
  resize();

  const particleChars = ["✦", "✧", "⋆", "❀", "♥", "·"];

  class Particle {
    constructor() {
      this.reset(true);
    }
    reset(initial = false) {
      this.x = Math.random() * width;
      this.y = initial ? Math.random() * height : height + 20;
      this.size = 9 + Math.random() * 14;
      this.speedY = 0.35 + Math.random() * 0.75;
      this.speedX = (Math.random() - 0.5) * 0.4;
      this.char = particleChars[Math.floor(Math.random() * particleChars.length)];
      this.alpha = 0.12 + Math.random() * 0.35;
      this.fade = 0.002 + Math.random() * 0.003;
      this.rot = Math.random() * Math.PI * 2;
      this.rotSpeed = (Math.random() - 0.5) * 0.02;
    }
    update() {
      this.y -= this.speedY;
      this.x += this.speedX;
      this.rot += this.rotSpeed;
      if (this.y < -30) this.reset();
    }
    draw() {
      ctx.save();
      ctx.translate(this.x, this.y);
      ctx.rotate(this.rot);
      ctx.fillStyle = `rgba(244, 182, 200, ${this.alpha})`;
      ctx.font = `${this.size}px serif`;
      ctx.fillText(this.char, 0, 0);
      ctx.restore();
    }
  }

  for (let i = 0; i < 35; i++) {
    particles.push(new Particle());
  }

  function loop() {
    ctx.clearRect(0, 0, width, height);
    particles.forEach(p => {
      p.update();
      p.draw();
    });
    requestAnimationFrame(loop);
  }
  loop();
}

// 9. BACKGROUND MUSIC CONTROLLER
// ==========================================================================
function initAudio() {
  if (BOOK.musicSrc) {
    bgAudio.src = BOOK.musicSrc;
  }

  musicBtn.addEventListener("click", () => {
    if (bgAudio.paused) {
      startMusic();
    } else {
      pauseMusic();
    }
  });
}

function startMusic() {
  bgAudio.play().then(() => {
    audioPlayed = true;
    musicBtn.classList.add("playing");
    musicBtn.title = "Pause music";
  }).catch(() => {
    // Autoplay policy blocked; requires direct click
  });
}

function pauseMusic() {
  bgAudio.pause();
  musicBtn.classList.remove("playing");
  musicBtn.title = "Play music";
}

// ==========================================================================
// 10. GESTURES & NAVIGATION LISTENERS
// ==========================================================================
function initNavigation() {
  // Arrow Buttons
  if (nextBtn) nextBtn.addEventListener("click", flipNext);
  if (prevBtn) prevBtn.addEventListener("click", flipPrev);

  // Click Left / Right Hit Zones
  if (hitNext) hitNext.addEventListener("click", flipNext);
  if (hitPrev) hitPrev.addEventListener("click", flipPrev);

  // Keyboard navigation
  window.addEventListener("keydown", e => {
    if (e.key === "ArrowRight" || e.key === " " || e.key === "PageDown") {
      e.preventDefault();
      flipNext();
    } else if (e.key === "ArrowLeft" || e.key === "PageUp") {
      e.preventDefault();
      flipPrev();
    }
  });

  // Touch Swipe Navigation (Mobile)
  let touchStartX = 0;
  let touchStartY = 0;
  const viewport = document.querySelector(".sketchbook-viewport");

  viewport.addEventListener("touchstart", e => {
    touchStartX = e.changedTouches[0].screenX;
    touchStartY = e.changedTouches[0].screenY;
  }, { passive: true });

  viewport.addEventListener("touchend", e => {
    const touchEndX = e.changedTouches[0].screenX;
    const touchEndY = e.changedTouches[0].screenY;
    const diffX = touchEndX - touchStartX;
    const diffY = touchEndY - touchStartY;

    // Minimum swipe threshold 40px and predominantly horizontal
    if (Math.abs(diffX) > 40 && Math.abs(diffX) > Math.abs(diffY)) {
      if (diffX < 0) {
        flipNext(); // Swipe Left -> Next
      } else {
        flipPrev(); // Swipe Right -> Prev
      }
    }
  }, { passive: true });
}
